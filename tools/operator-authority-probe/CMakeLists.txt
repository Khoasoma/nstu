cmake_minimum_required(VERSION 3.25)
project(nstu_operator_authority_probe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Operator authority qualification requires native Windows")
endif()
option(NSTU_ENABLE_WERROR "Treat warnings as errors" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/NstuWarnings.cmake")
add_library(nstu_windows_operator_authorization STATIC ../../server/src/windows_operator_authorization.cpp)
target_include_directories(nstu_windows_operator_authorization PUBLIC ../../server/include)
target_compile_definitions(nstu_windows_operator_authorization PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
target_link_libraries(nstu_windows_operator_authorization PUBLIC advapi32 netapi32 wtsapi32)
nstu_enable_warnings(nstu_windows_operator_authorization)
add_executable(nstu_operator_authority_tests tests.cpp)
target_link_libraries(nstu_operator_authority_tests PRIVATE nstu_windows_operator_authorization)
if(MSVC)
    target_compile_options(nstu_operator_authority_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(nstu_operator_authority_tests PRIVATE -UNDEBUG)
endif()
nstu_enable_warnings(nstu_operator_authority_tests)
enable_testing()
add_test(NAME auth01.injected_unit COMMAND nstu_operator_authority_tests --unit)
set_tests_properties(auth01.injected_unit PROPERTIES LABELS "injected;unit" TIMEOUT 30)
add_test(NAME auth01.native_read_only COMMAND nstu_operator_authority_tests --native-read-only)
set_tests_properties(auth01.native_read_only PROPERTIES LABELS "native;read-only;no-managed-grants" TIMEOUT 30)

add_test(NAME auth01.native_owned_restrictions COMMAND nstu_operator_authority_tests --native-owned-restrictions)
set_tests_properties(auth01.native_owned_restrictions PROPERTIES LABELS "native;owned-token;no-managed-grants" TIMEOUT 30)
add_test(NAME auth01.native_query_errors COMMAND nstu_operator_authority_tests --native-query-errors)
set_tests_properties(auth01.native_query_errors PROPERTIES LABELS "injected;native-adapter;no-managed-grants" TIMEOUT 30)
