cmake_minimum_required(VERSION 3.20)
project(nstu_file_output_probe LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
    message(FATAL_ERROR "The file output probe requires Windows.")
endif()
option(NSTU_ENABLE_WERROR "Treat compiler warnings as errors" ON)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/NstuWarnings.cmake")
function(nstu_enable_test_assertions target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endfunction()
enable_testing()
add_executable(nstu_checked_file_output_tests
    "${CMAKE_CURRENT_LIST_DIR}/../../tests/checked_file_output_tests.cpp")
target_include_directories(nstu_checked_file_output_tests PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/../../server/include")
target_compile_features(nstu_checked_file_output_tests PRIVATE cxx_std_20)
target_compile_definitions(nstu_checked_file_output_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(nstu_checked_file_output_tests PRIVATE shell32 ole32)
if(MSVC)
    target_compile_options(nstu_checked_file_output_tests PRIVATE /utf-8)
endif()
nstu_enable_warnings(nstu_checked_file_output_tests)
nstu_enable_test_assertions(nstu_checked_file_output_tests)
add_test(NAME nstu.checked_file_output COMMAND nstu_checked_file_output_tests)
